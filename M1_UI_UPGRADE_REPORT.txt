```javascript
async function importCharacterPresets() {
  if (!outputFolder) {
    alert("Vui lòng chọn hoặc mở project trước");
    setStatus("Hủy import nhân vật: Chưa chọn project folder", "error");
    return;
  }
  try {
    setStatus("Đang chọn và import presets nhân vật...", "running");
    const result = await window.videoPlannerAPI.importCharacterPresets(outputFolder);
    if (result && result.ok) {
      setStatus(`Đã import thành công ${result.copiedFiles.length} file nhân vật.`, "ok");
    } else {
      setStatus("Import nhân vật đã bị hủy bởi người dùng.", "idle");
    }
  } catch (error) {
    setStatus(`Lỗi import nhân vật: ${error.message}`, "error");
    alert(`Lỗi import nhân vật: ${error.message}`);
  }
}
```

```javascript
async function importCharacterPresetsHandler(_event, projectPath) {
  if (!projectPath) {
    throw new Error('Thiếu projectPath để import nhân vật.');
  }

  const result = await dialog.showOpenDialog({
    title: 'Chọn các file nhân vật (Presets)',
    properties: ['openFile', 'multiSelections'],
    filters: [
      { name: 'Presets (*.png, *.jpg, *.jpeg, *.txt)', extensions: ['png', 'jpg', 'jpeg', 'txt'] },
      { name: 'Images (*.png, *.jpg, *.jpeg)', extensions: ['png', 'jpg', 'jpeg'] },
      { name: 'Descriptions (*.txt)', extensions: ['txt'] },
      { name: 'All files', extensions: ['*'] }
    ]
  });

  if (result.canceled || !result.filePaths || result.filePaths.length === 0) {
    return { ok: false, canceled: true };
  }

  const targetDir = path.join(projectPath, 'characters');
  const fsSync = require('fs');
  fsSync.mkdirSync(targetDir, { recursive: true });

  const copiedFiles = [];
  for (const srcPath of result.filePaths) {
    const fileName = path.basename(srcPath);
    const destPath = path.join(targetDir, fileName);
    fsSync.copyFileSync(srcPath, destPath);
    copiedFiles.push(fileName);
  }

  // Open the characters directory in OS file explorer
  await shell.openPath(targetDir);

  await appendAppLog(null, {
    source: 'main',
    kind: 'ok',
    text: `Đã import ${copiedFiles.length} file nhân vật vào folder: ${targetDir}`,
    details: { copiedFiles }
  }).catch(() => null);

  return { ok: true, copiedFiles };
}
```
